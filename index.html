Integrasi Google Apps Script (Backend Pencatat Pelanggaran & Host Web App)

Salin kode berikut ke dalam proyek Google Apps Script Anda (script.google.com) untuk menghosting aplikasi ExamBro ini langsung dari server Google dan mencatat log kecurangan siswa secara otomatis ke Google Sheets.

Langkah 1: Buat Project Buka Google Drive > Baru > Lainnya > Google Apps Script. Buat file Code.gs dan Index.html.
Langkah 2: Tempel Kode Tempel kode GS di bawah ke Code.gs, dan seluruh kode halaman ini ke Index.html.
Langkah 3: Deploy Web App Klik Deploy > New Deployment > Web App. Pilih akses "Anyone" lalu bagikan linknya ke siswa.
Code.gs (Google Apps Script - Support Google Sites) 
/**
 * BACKEND GOOGLE APPS SCRIPT UNTUK CBT MTs. AL HASAN MAARIF
 * Support Google Sites Embed (Izin Kamera & Fullscreen) + Pencatat Log Pelanggaran
 */

// 1. Fungsi utama untuk menampilkan halaman Web App ExamBro di Google Sites
function doGet(e) {
  return HtmlService.createHtmlOutputFromFile('Index')
    .setTitle('CBT MTs. Al Hasan Maarif')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
    .setSandboxMode(HtmlService.SandboxMode.IFRAME)
    .addMetaTag('viewport', 'width=device-width, initial-scale=1, maximum-scale=1, user-scalable=no');
}

// 2. Fungsi untuk mencatat aktivitas/pelanggaran siswa ke Google Sheets secara real-time
function logViolationToSheet(data) {
  try {
    // OPSIONAL: Masukkan ID Google Spreadsheet jika ingin mencatat log kecurangan ke Sheets
    const SPREADSHEET_ID = 'MASUKKAN_ID_SPREADSHEET_ANDA_DISINI'; 
    if (SPREADSHEET_ID === 'MASUKKAN_ID_SPREADSHEET_ANDA_DISINI') {
      return { success: true, message: 'Mode tanpa spreadsheet aktif' };
    }
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    let sheet = ss.getSheetByName('Log_Keamanan_Ujian');
    
    // Buat sheet otomatis jika belum ada
    if (!sheet) {
      sheet = ss.insertSheet('Log_Keamanan_Ujian');
      sheet.appendRow(['Waktu Kejadian', 'Perangkat / Sesi', 'Jenis Pelanggaran', 'Pelanggaran Ke-', 'Status']);
      sheet.getRange('A1:E1').setFontWeight('bold').setBackground('#1e293b').setFontColor('#ffffff');
    }
    
    // Simpan baris log baru
    sheet.appendRow([
      new Date(),
      data.sessionInfo || 'Peserta CBT',
      data.reason,
      data.count,
      data.status
    ]);
    
    return { success: true, message: 'Log tercatat di server' };
  } catch (error) {
    return { success: false, message: error.toString() };
  }
}
